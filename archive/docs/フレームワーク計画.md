## 1. 前提と制約

### ユーザー要件
- **バイブコーディング対応**: AI（Claude Code）がソースを直接編集して開発を完結させる
- **ゲーム制作者はコードに触れない**: 将来的にWebベースのエディタUIを提供予定（今回スコープ外）
- **ルールエンジンは描画から完全独立**: packages/engine は DOM/Canvas に一切依存しない
- **Webブラウザで描画**: Unity は使用しない
- **PuzzleScript パーサーはスコープ外**: ルールは JSON で直接定義
- **ビルドして配布可能**: 単一 HTML ファイルとしてゲームをエクスポート

### 設計制約（設計思想.md より）
- グリッドベースのみ（自由座標なし）
- ターンベースのみ（リアルタイム要素なし）
- 整数のみ（float 禁止、Bool と Int のみ）
- イミュータブル状態（GameState は常に新規生成）
- 原子的ルール適用（中間状態は観測不能）
- 決定論的実行（同一入力 + 同一状態 = 同一結果）

## 2. 技術スタック

| 項目 | 選定 | バージョン | 理由 |
|:---|:---|:---|:---|
| 言語 | TypeScript | 5.x (strict mode) | ブラウザネイティブ、JSON 親和性、`Readonly<T>` によるイミュータブル強制 |
| ランタイム | Node.js | 20+ LTS | 開発・テスト・ビルド用 |
| パッケージ管理 | pnpm | 9+ | workspaces によるモノレポ管理、厳格な依存関係 |
| ライブラリビルド | tsup | 最新 | TypeScript ライブラリの高速バンドル |
| アプリビルド | Vite | 6+ | 高速 HMR、開発体験 |
| テスト | Vitest | 最新 | TypeScript ネイティブ対応、Vite 統合 |
| レンダリング | HTML5 Canvas 2D | — | 外部依存なし、グリッドベースに十分 |
| ゲーム配布 | vite-plugin-singlefile | 最新 | 単一 HTML でオフライン動作 |

### 5.2 types/game-state.ts

```typescript
/**
 * オブジェクトインスタンスの文字列表現。
 * フォーマット: "Name" | "Name:tag1" | "Name:tag1:tag2"
 * 例: "Player", "Player:right", "Box:on_goal"
 */
export type ObjectInstance = string;

/** 1つのセルに存在するオブジェクトのリスト（イミュータブル） */
export type Cell = readonly ObjectInstance[];

/**
 * 2次元グリッド。puzzle[y][x] でアクセス。
 * y=0 が最上行、x=0 が最左列。
 */
export type Puzzle = readonly (readonly Cell[])[];

/** グローバル変数の値（int, bool, string のみ） */
export type GlobalValue = number | boolean | string;

/** グローバル状態のキーバリューストア */
export type GlobalState = Readonly<Record<string, GlobalValue>>;

/** ゲーム全体の状態（完全にイミュータブル） */
export interface GameState {
  readonly globalState: GlobalState;
  readonly puzzle: Puzzle;
  readonly width: number;   // グリッドの幅（列数）
  readonly height: number;  // グリッドの高さ（行数）
}
```

### 5.3 types/object.ts

```typescript
/**
 * オブジェクトの定義（マスターデータ）。
 * ゲーム中に変化しない静的な情報。
 */
export interface ObjectDefinition {
  /** 一意な名称。"Player", "Wall", "Box" 等 */
  readonly name: string;

  /**
   * 描画レイヤー ID。数値が大きいほど手前に描画される。
   * 0: 背景レイヤー（Floor, Goal 等）
   * 1: オブジェクトレイヤー（Player, Wall, Box 等）
   */
  readonly layerId: number;

  /**
   * 抽象プロパティのリスト。ルール内で @プレフィックスで参照可能。
   * 例: ["Movable", "Solid"]
   */
  readonly properties: readonly string[];

  /**
   * タグのカテゴリと取りうる値の定義。
   * キー: タグカテゴリ名、値: そのカテゴリの全有効値
   * 例: { "direction": ["up", "down", "left", "right"], "state": ["on_goal"] }
   */
  readonly tags: Readonly<Record<string, readonly string[]>>;

  /**
   * デフォルトタグ（省略可）。
   * レベルデータで "Player" と書いたとき、自動付与されるタグ。
   * 例: ["down"] → "Player:down" として初期化される
   */
  readonly defaultTags?: readonly string[];

  /**
   * スプライト/描画設定（省略可）。
   * レンダラーが参照する。エンジンは無視する。
   */
  readonly sprite?: {
    /** 単色表示時の色（CSS カラー文字列） */
    readonly color?: string;
    /** タグごとの色バリエーション */
    readonly colorByTag?: Readonly<Record<string, string>>;
  };
}
```

### 5.4 types/rule.ts

```typescript
import type { Direction } from "./input.js";

/** ルールの方向指定 */
export type RuleDirection =
  | "none"       // 方向なし（1x1パターンや方向非依存）
  | Direction    // 特定の1方向: "up" | "down" | "left" | "right"
  | "vertical"   // 上下の2方向
  | "horizontal" // 左右の2方向
  | "any";       // 4方向すべて（自動的に4バリアント生成）

/** ルールの適用方式 */
export type ApplicationMode = "once" | "until_stable";

/**
 * パターン内の1セルの条件。
 * before ブロック内のセルに指定する。
 */
export interface PatternCell {
  /**
   * 置換対象オブジェクト。
   * このセルに指定オブジェクトが存在することを要求し、
   * after ブロックの対応セルのオブジェクトに置換される。
   * "@Movable" のようにプロパティ指定可能。
   * "$color" のようにタグバインディング可能。
   */
  readonly objects?: readonly string[];

  /**
   * 存在確認のみ。マッチに参加するが置換対象にはならない。
   * セルにこのオブジェクトが存在していることを要求する。
   */
  readonly hasObjects?: readonly string[];

  /**
   * 不存在確認。セルにこのオブジェクトが存在しないことを要求する。
   * 例: "Wall" が存在しないセルにのみマッチ
   */
  readonly noObjects?: readonly string[];
}

/**
 * after ブロック内の1セルの指定。
 * before の objects に対応する置換先を記述する。
 */
export interface AfterPatternCell {
  /**
   * 置換先オブジェクト。
   * before の objects と1対1で対応する。
   * 空配列 [] は「オブジェクトを消す」を意味する。
   * "$color" タグは before でバインドされた値に展開される。
   */
  readonly objects?: readonly string[];
}

/**
 * 1つのパターンブロック（before/after ペア）。
 * before は2Dパターンで、方向指定によって回転される。
 */
export interface PatternBlock {
  /** マッチ条件（2D配列: [row][col]）*/
  readonly before: readonly (readonly PatternCell[])[];
  /** 置換先（2D配列: [row][col]。before と同じサイズ）*/
  readonly after: readonly (readonly AfterPatternCell[])[];
}

/** 比較演算子 */
export type ComparisonOp = "==" | "!=" | "<" | "<=" | ">" | ">=";

/** 条件式（グローバル変数に対する条件） */
export interface Condition {
  /** グローバル変数名 */
  readonly variable: string;
  /** 比較演算子 */
  readonly op: ComparisonOp;
  /** 比較対象の値 */
  readonly value: number | boolean | string;
}

/** エフェクト: グローバル変数の設定 */
export interface SetEffect {
  readonly type: "set";
  readonly variable: string;
  readonly value: number | boolean | string;
}

/** エフェクト: グローバル変数の加減算 */
export interface ChangeEffect {
  readonly type: "change";
  readonly variable: string;
  readonly amount: number;
}

/** エフェクト: サウンド再生（レンダラーに委譲） */
export interface SoundEffect {
  readonly type: "sound";
  readonly soundId: string;
}

/** エフェクト: メッセージ表示（レンダラーに委譲） */
export interface MessageEffect {
  readonly type: "message";
  readonly text: string;
}

/** エフェクト: 他のルールグループを呼び出す */
export interface CallEffect {
  readonly type: "call";
  readonly groupName: string;
}

/** エフェクトの統合型 */
export type Effect =
  | SetEffect
  | ChangeEffect
  | SoundEffect
  | MessageEffect
  | CallEffect;

/** 1つのルール定義 */
export interface Rule {
  /** ルール名（デバッグ・ログ用、省略可） */
  readonly name?: string;

  /** パターンの方向指定 */
  readonly direction: RuleDirection;

  /**
   * 個別ルールの適用方式。
   * "once": 最初の1マッチのみ適用
   * "until_stable": 変化がなくなるまで繰り返し適用
   * 省略時のデフォルト: "once"
   */
  readonly application?: ApplicationMode;

  /**
   * パターンブロックの配列。
   * 複数ブロックがある場合、すべてがマッチしたときのみ全ブロックが同時適用される。
   */
  readonly patterns: readonly PatternBlock[];

  /** 適用前に評価される条件（すべて AND）。省略時は常に真 */
  readonly conditions?: readonly Condition[];

  /** 適用後に実行されるエフェクト。省略時はなし */
  readonly effects?: readonly Effect[];
}

/** ルールグループ */
export interface RuleGroup {
  /** グループ名（call エフェクトで参照する際のキー） */
  readonly name: string;

  /**
   * グループの適用方式。
   * "once": グループ内の全ルールを1回ずつ順に実行
   * "until_stable": グループ全体で変化がなくなるまで繰り返し
   */
  readonly application: ApplicationMode;

  /** グループ内のルール（配列順に実行） */
  readonly rules: readonly Rule[];
}
```

### 5.5 types/level.ts

```typescript
import type { GlobalValue } from "./game-state.js";

/** レベルの凡例: 1文字 → オブジェクト名の配列 */
export type Legend = Readonly<Record<string, readonly string[]>>;

/**
 * クリア条件。
 * 複数指定時は AND（すべて満たす必要がある）。
 */
export type WinCondition =
  | {
      /** "all": 指定オブジェクトが全て指定条件を満たす */
      readonly type: "all";
      /** 対象オブジェクト名 */
      readonly object: string;
      /** 同じセルに存在すべきオブジェクト名 */
      readonly on: string;
    }
  | {
      /** "no": 指定オブジェクトが盤面に1つも存在しない */
      readonly type: "no";
      readonly object: string;
    }
  | {
      /** "some": 指定オブジェクトが盤面に1つ以上存在する */
      readonly type: "some";
      readonly object: string;
      readonly on: string;
    }
  | {
      /** "global": グローバル変数の条件 */
      readonly type: "global";
      readonly variable: string;
      readonly op: "==" | "!=" | "<" | "<=" | ">" | ">=";
      readonly value: number | boolean | string;
    };

/** レベルデータ */
export interface LevelData {
  /** レベル名（表示用、省略可） */
  readonly name?: string;
```

```typescript

  readonly blockIndex: number;
  /** マッチした左上の座標 */
  readonly x: number;
  readonly y: number;
  /** マッチした方向 */
  readonly direction: Direction | null;
  /** タグバインディング */
  readonly bindings: Readonly<Record<string, string>>;
  /**
   * マッチした各セルの情報（置換に必要）。
   * 配列のインデックスはパターン内の位置に対応。
   */
  readonly matchedCells: readonly MatchedCellInfo[];
}

/** セルレベルのマッチ情報 */
export interface MatchedCellInfo {
  /** グリッド上の実座標 */
  readonly x: number;
  readonly y: number;
  /**
   * objects で指定されたパターンと実際にマッチしたオブジェクトの対応。
   * キー: パターン文字列, 値: 実際のオブジェクトインスタンス

 *   left:  180度回転
```

### 7.9 core/game-manager.ts

```typescript
import type { InputEvent, GameState, GameData, LevelData, Effect } from "../types/index.js";
import { ObjectDB } from "./object-db.js";
import { HistoryManager } from "./history-manager.js";
import { loadLevel } from "./level-loader.js";
import { processTurn } from "./rule-processor.js";

/** ゲームのフェーズ */
export type GamePhase = "playing" | "won" | "paused";

/** ターン処理の結果（レンダラーに渡す情報を含む） */
export interface TurnResult {
  readonly state: GameState;
  readonly phase: GamePhase;
  readonly effects: readonly Effect[];
  readonly changed: boolean;
}

export class GameManager {
  private currentState: GameState;
  private readonly objectDB: ObjectDB;
  private readonly history: HistoryManager;
  private readonly gameData: GameData;
  private phase: GamePhase;
  private initialState: GameState;  // restart 用

  constructor(gameData: GameData);

  /**
   * レベルをロードして初期状態を設定する。
   * 履歴をクリアし、初期状態をプッシュする。
   */
  loadLevel(levelData: LevelData): GameState;

  /**
   * 入力イベントを処理する。
   *
   * 処理フロー:
   * 1. phase が "playing" でなければ無視
   * 2. InputEvent の type に応じて分岐:
   *    - "undo": history.undo() を実行
   *    - "redo": history.redo() を実行
   *    - "restart": initialState を history.push して復帰
   *    - "move" / "action" / "wait":
   *      a. 入力に基づいて PlayerControlled オブジェクトの移動意図をセットアップ
   *         ※ 移動は「Player の隣セルへの移動」をルールに委ねるため、
   *            ここでは入力方向をグローバル変数 `_input_direction` にセットするだけ
   *      b. processTurn を呼び出してルール処理
   *      c. 結果の GameState を history.push
   *      d. winConditions を評価
   * 3. TurnResult を返す
   */
  processInput(input: InputEvent): TurnResult;

  /**
   * クリア条件を評価する。
   * 全 WinCondition が満たされたら "won" を返す。
   */
  private checkWinConditions(
    state: GameState,
    winConditions: readonly WinCondition[]
  ): boolean;

  /**
   * 現在の GameState を返す（読み取り専用）。
   */
  getState(): GameState;

  /**
   * 現在の GamePhase を返す。
   */
  getPhase(): GamePhase;
}
```

**入力処理の重要な設計判断**:

プレイヤーの移動は **ルールとして定義する**。GameManager は入力方向をグローバル変数 `_input_direction` に文字列としてセットし（`"up"`, `"down"`, `"left"`, `"right"`, `"none"`）、ルールが `_input_direction` を参照してプレイヤーを移動させる。これにより移動ロジックもデータ駆動になる。

`_input_direction` の処理:
1. `"move"` 入力: `_input_direction` を入力方向にセット
2. `"action"` 入力: `_input_direction` を `"none"` にセット、`_action` を `true` にセット
3. `"wait"` 入力: `_input_direction` を `"none"` にセット
4. ルール処理実行後: `_input_direction` と `_action` をリセット

## 8. アルゴリズム詳細

### 8.1 パターンマッチングアルゴリズム

これはフレームワーク最大の複雑さを持つ部分。以下に疑似コードで詳述する。

```
function findMatch(puzzle, block, ruleDirection, objectDB, existingBindings):
    // 1. 方向リストを生成
    directions = expandDirections(ruleDirection)
    // "none" → [null]
    // "up" → ["up"]
    // "vertical" → ["up", "down"]
    // "horizontal" → ["left", "right"]
    // "any" → ["up", "right", "down", "left"]

    for each direction in directions:
        // 2. パターンを方向に応じて回転
        rotatedBefore = rotatePattern(block.before, direction)

        patternHeight = rotatedBefore.length
        patternWidth = rotatedBefore[0].length

        // 3. グリッドをスキャン（左上→右下）
        for y = 0 to puzzle.height - patternHeight:
            for x = 0 to puzzle.width - patternWidth:
                // 4. この位置でパターンがマッチするか試行
                result = tryMatch(puzzle, rotatedBefore, x, y, direction, objectDB, existingBindings)
                if result != null:
                    return result

    return null  // マッチなし


function tryMatch(puzzle, pattern, startX, startY, direction, objectDB, existingBindings):
    bindings = copy(existingBindings ?? {})
    matchedCells = []

    for py = 0 to pattern.length - 1:
        for px = 0 to pattern[py].length - 1:
            cellPattern = pattern[py][px]
            puzzleX = startX + px
            puzzleY = startY + py
            cell = puzzle[puzzleY][puzzleX]

            // objects チェック（置換対象）
            if cellPattern.objects:
                for each patternObj in cellPattern.objects:
                    // @プロパティの展開
                    candidates = objectDB.resolvePattern(patternObj の名前部分)
                    matched = false
                    for each candidateName in candidates:
                        fullPattern = candidateName + patternObj のタグ部分
                        for each actualObj in cell:
                            newBindings = matchObjectPattern(actualObj, fullPattern, bindings)
                            if newBindings != null:
                                bindings = newBindings
                                記録: patternObj → actualObj
                                matched = true
                                break
                        if matched: break
                    if not matched: return null  // マッチ失敗

            // hasObjects チェック（存在確認のみ）
            if cellPattern.hasObjects:
                for each requiredObj in cellPattern.hasObjects:
                    candidates = objectDB.resolvePattern(requiredObj の名前部分)
                    found = false
                    for each candidateName in candidates:
                        for each actualObj in cell:
                            if actualObj の名前 == candidateName:
                                found = true; break
                        if found: break
                    if not found: return null

            // noObjects チェック（不存在確認）
            if cellPattern.noObjects:
                for each forbiddenObj in cellPattern.noObjects:
                    candidates = objectDB.resolvePattern(forbiddenObj の名前部分)
                    for each candidateName in candidates:
                        for each actualObj in cell:
                            if actualObj の名前 == candidateName:
                                return null  // 存在してはいけないものが存在

            matchedCells.push({ x: puzzleX, y: puzzleY, objectMatches: ... })

    return { blockIndex, x: startX, y: startY, direction, bindings, matchedCells }
```

### 8.2 パターン回転アルゴリズム

基準方向は **right**（パターンは右向きに記述されている）。

```
rotatePattern(pattern, direction):
    switch direction:
        case "right" or null:
            return pattern  // そのまま
        case "down":
            // 90度時計回り: [row][col] → [col][maxRow - row]
            rows = pattern.length
            cols = pattern[0].length
            new[cols][rows]
            for r=0..rows-1, c=0..cols-1:
                new[c][rows-1-r] = pattern[r][c]
            return new
        case "left":
            // 180度回転: [row][col] → [maxRow-row][maxCol-col]
            rows = pattern.length
            cols = pattern[0].length
            new[rows][cols]
            for r=0..rows-1, c=0..cols-1:
                new[rows-1-r][cols-1-c] = pattern[r][c]
            return new
        case "up":
            // 90度反時計回り: [row][col] → [maxCol-col][row]
            rows = pattern.length
            cols = pattern[0].length
            new[cols][rows]
            for r=0..rows-1, c=0..cols-1:
                new[cols-1-c][r] = pattern[r][c]
            return new
```

**相対方向タグの変換**:
回転時に、パターン内の相対方向タグ (`>`, `<`, `^`, `v`) も同じ回転を適用する。

| 元のタグ | right (0°) | down (90°) | left (180°) | up (270°) |
|:---|:---|:---|:---|:---|
| `>` (右) | `>` | `v` | `<` | `^` |
| `<` (左) | `<` | `^` | `>` | `v` |
| `^` (上) | `^` | `>` | `v` | `<` |
| `v` (下) | `v` | `<` | `^` | `>` |

### 8.3 ルール適用フロー（1ルール）

```
function applyRule(state, rule, objectDB):
    // 1. 条件チェック
    if rule.conditions:
        for each cond in rule.conditions:
            if not evaluateCondition(cond, state.globalState):
                return { state, changed: false, effects: [] }

    application = rule.application ?? "once"

    if application == "once":
        return applyRuleOnce(state, rule, objectDB)
    else: // "until_stable"
        totalEffects = []
        loop:
            result = applyRuleOnce(state, rule, objectDB)
            totalEffects.push(...result.effects)
            if not result.changed:
                return { state, changed: totalEffects.length > 0, effects: totalEffects }
            state = result.state

function applyRuleOnce(state, rule, objectDB):
    // 複数パターンブロックの処理
    allMatches = []
    bindings = {}

    for each block in rule.patterns:
        match = findMatch(state.puzzle, block, rule.direction, objectDB, bindings)
        if match == null:
            return { state, changed: false, effects: [] }  // 1つでもマッチしなければ不適用
        allMatches.push(match)
        bindings = { ...bindings, ...match.bindings }

    // 全ブロックがマッチ → 原子的に適用
    newPuzzle = state.puzzle
    for each (match, blockIndex) in allMatches:
        afterPattern = rotatePattern(rule.patterns[blockIndex].after, match.direction)
        newPuzzle = applyMatch(newPuzzle, match, afterPattern, match.direction, bindings)

    // エフェクト適用
    { globalState, pendingEffects } = applyEffects(state.globalState, rule.effects ?? [])

    newState = { ...state, puzzle: newPuzzle, globalState }
    return { state: newState, changed: true, effects: pendingEffects }
```

### 8.4 ルールグループ実行フロー

```
function applyRuleGroup(state, group, allGroups, objectDB):
    allEffects = []

    if group.application == "once":
        for each rule in group.rules:
            result = applyRule(state, rule, objectDB)
            state = result.state
            allEffects.push(...result.effects)

            // call エフェクトの処理
            for each effect in result.effects:
                if effect.type == "call":
                    targetGroup = allGroups.find(g => g.name == effect.groupName)
                    callResult = applyRuleGroup(state, targetGroup, allGroups, objectDB)
                    state = callResult.state
                    allEffects.push(...callResult.effects)

        return { state, changed: allEffects.length > 0, effects: allEffects }

    else: // "until_stable"
        loop:
            groupChanged = false
            for each rule in group.rules:
                result = applyRule(state, rule, objectDB)
                if result.changed:
                    groupChanged = true
                state = result.state
                allEffects.push(...result.effects)
                // call エフェクトも処理（上記と同様）
            if not groupChanged:
                return { state, changed: allEffects.length > 0, effects: allEffects }
```

### 8.5 クリア条件判定アルゴリズム

```
function checkWinConditions(state, winConditions):
    for each condition in winConditions:
        switch condition.type:
            case "all":
                // 指定オブジェクトが全て指定オブジェクトと同じセルにある
                allPositions = findAllObjects(state.puzzle, condition.object)
                if allPositions.length == 0: return false  // 0個は未クリア
                for each pos in allPositions:
                    cell = getCell(state.puzzle, pos.x, pos.y)
                    if not cell.any(obj => getObjectName(obj) == condition.on):
                        return false

            case "no":
                // 指定オブジェクトが盤面に存在しない
                positions = findAllObjects(state.puzzle, condition.object)
                if positions.length > 0: return false

            case "some":
                // 指定オブジェクトが1つ以上指定オブジェクトと同じセルにある
                allPositions = findAllObjects(state.puzzle, condition.object)
                found = false
                for each pos in allPositions:
                    cell = getCell(state.puzzle, pos.x, pos.y)
                    if cell.any(obj => getObjectName(obj) == condition.on):
                        found = true; break
                if not found: return false

            case "global":
                if not evaluateCondition(condition, state.globalState): return false

    return true  // 全条件クリア
```

```
- canUndo / canRedo の正確性
```
